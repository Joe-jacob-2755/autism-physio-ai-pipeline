Module 2A — Data Simulation (7 Steps)

flowchart TD

    HEAD[/"MODULE 2A — Data Simulation v1.1.0"\]:::head
    HEAD --> INPUT[/"Inputs:\nduration_s, n_events,\nevent_duration_s, emotions[],\nnoise_level, seed, n_users"/]:::data

    INPUT --> S1["Step 1: UserProfileGenerator\nDraw from POPULATION distributions\n(age 5-15, gender 75/22/3%,\nseverity Low/Med/Severe,\nverbal 55/25/20%)"]:::proc
    S1 --> O1[/"UserProfile objects\n- Baseline physiology params\n- demographic_dict()"/]:::out

    O1 --> S2["Step 2: EventScheduler\nPlace non-overlapping events\n(min_gap_s between events,\nmin_lead_s before first event)"]:::proc
    S2 --> O2[/"EventConfig list\n(emotion, start_s, end_s,\nevent_id, duration_s)"/]:::out

    O2 --> S3["Step 3: Baseline Signal Generation"]:::proc
    subgraph S3_DETAIL ["Signal Models"]
        direction LR
        S3A["EDA: SCL tonic\n(drift + respiration)"]:::sub
        S3B["BVP: 3-Gaussian\nPPG morphology"]:::sub
        S3C["IBI: AR(1) HRV\nmodel"]:::sub
        S3D["ST: thermal\ndrift model"]:::sub
        S3E["ACC: gravity +\nbreathing artefact"]:::sub
    end
    S3 --> O3[/"Raw baseline signals\nEDA@4Hz, BVP@64Hz,\nIBI@event, ST@4Hz,\nACC@32Hz x 3 axes"/]:::out

    O3 --> S4["Step 4: Event Modulation\nApply 10 emotion profiles\n(valence-arousal mapped)\nSeverity scaling:\nLow=1.0x, Med=1.3x, Severe=1.6x\nAge effect: +1.2 bpm/yr younger"]:::proc
    S4 --> O4[/"Modulated signals\n(emotion-specific patterns\nin event windows)"/]:::out

    O4 --> S5["Step 5: NoiseInjector\nEDA: drift + powerline (50Hz)\nBVP: motion + baseline wander\nACC: correlated cross-axis vibration\nTiers: low / medium / high"]:::proc
    S5 --> O5[/"Noisy signals"/]:::out

    O5 --> S6["Step 6: Clip & Validate\nEnforce physiological ranges:\nEDA: 0.01-30 uS\nBVP: -300 to +300 nT\nIBI: 300-1500 ms\nST: 25-40 C | ACC: -4 to +4 g"]:::proc
    S6 --> O6[/"Validated signals"/]:::out

    O6 --> S7["Step 7: AutoAnnotator\n- Event-level labels\n- SQI (0-1) per 10s window\n- Sample-level target_label\n- Baseline window markers"]:::proc
    S7 --> OUTPUT

    OUTPUT[/"OUTPUTS --> M1:\nEDA.csv, BVP.csv, IBI.csv,\nST.csv, ACC.csv\ncombined_signals.csv (64 Hz)\nannotations_events.csv\nannotations_signal_quality.csv\nannotations_sample_labels.csv\nmetadata.json\nsignal_*.png"/]:::out

    HEAD[/"MODULE 3 — Data Preprocessing v1.0.0"\]:::head
    HEAD --> INPUT[/"Inputs:\nPipelinePacket | folder path |\ndict of DataFrames\n+ demographics"/]:::data

    INPUT --> S1["Step 1: SignalCleaner"]:::proc
    subgraph S1D ["Cleaning Rules"]
        direction LR
        S1A["Missing > 70%\n--> DISCARD channel"]:::sub
        S1B["Missing < 70%\n--> linear interpolation"]:::sub
        S1C["Out-of-range\n--> NaN --> re-fill"]:::sub
        S1D2["Flatline detection\n(> 5s constant)"]:::sub
    end
    S1 --> O1[/"CleaningReport per channel\n(n_missing, n_oor, n_flat,\nn_discarded, n_interpolated)"/]:::out

    O1 --> S2["Step 2: SignalFilterManager"]:::proc
    subgraph S2D ["Two-Stage Filtering"]
        direction LR
        S2A["Stage A: Hampel filter\n(sliding MAD, optional)\nRemove impulse artefacts"]:::sub
        S2B["Stage B: Main filter"]:::sub
    end

    S2 --> FTYPE{{"Filter Type?"}}:::dec
    FTYPE -->|"butterworth"| BW["Butterworth zero-phase\nEDA: LP 1.0 Hz\nBVP: BP 0.5-8.0 Hz\nST: LP 0.1 Hz\nACC: BP 0.1-15.0 Hz"]:::sub
    FTYPE -->|"kalman"| KF["Kalman RTS smoother\n(optimal for EDA/ST)"]:::sub

    BW --> O2[/"Filtered signals"/]:::out
    KF --> O2

    O2 --> S3["Step 3: FeatureExtractor\n60s window, 50% overlap\n80 features per window\n(18 EDA + 13 BVP + 17 IBI +\n10 ST + 14 ACC + 8 ACC variants)"]:::proc
    S3 --> O3[/"Per-signal feature DataFrames\n(one row per window)"/]:::out

    O3 --> S4["Step 4: DemographicEncoder\nOrdinal encoding:\nage (continuous)\ngender: M=0, F=1, NB=2\nseverity: L=1, M=2, S=3\nverbal: V=0, MV=1, NV=2\ncomorbidity: No=0, Yes=1"]:::proc
    S4 --> O4[/"Feature DFs +\ndemographic columns appended"/]:::out

    O4 --> S5["Step 5: FeatureNormaliser\nRobustScaler (median + IQR)\n+ FeatureFuser\n(merge_asof, 30s tolerance,\nwindow alignment across signals)"]:::proc
    S5 --> O5[/"Per-signal (raw + norm)\nCombined (raw + norm)\nFitted scaler object"/]:::out

    O5 --> S6["Step 6: Visualiser + Exporter"]:::proc
    subgraph S6D ["Output Generation"]
        direction LR
        S6A["Processed signal plots\n(with event timeline bar)"]:::sub
        S6B["Raw vs processed\ncomparison plots\n(SNR gain annotation)"]:::sub
        S6C["CSV export\n(4 sets: raw/norm x\nindividual/combined)"]:::sub
    end

    S6 --> OUTPUT

    OUTPUT[/"OUTPUTS --> M4 + M5 Analyser:\n--- features_raw/ ---\nEDA_features.csv, BVP_features.csv\nIBI_features.csv, ST_features.csv\nACC_features.csv\ncombined_features.csv\n--- features_normalised/ ---\n*_features_norm.csv\ncombined_features_norm.csv\n--- Plots ---\nprocessed_*.png, comparison_*.png\n--- Reports ---\ncleaning_report.csv\npreprocessing_metadata.json"/]:::out

    HEAD[/"MODULE 4 — Feature Engineering v1.0.0"\]:::head

    INPUT --> S1["Step 1: FeatureExtractor\n93 features total (windowed)\nEDA: 22 features\nBVP: 16 features\nIBI: 17 features\nST: 12 features\nACC: 26 features"]:::proc

    subgraph S1D ["Feature Categories"]
        direction LR
        S1A["Level 1: Derived\nSCL/SCR, beat morphology,\nHRV time-domain, SVM,\nthermal delta"]:::sub
        S1B["Level 2: Time domain\nmean, std, skew, kurtosis,\nrange, ZCR"]:::sub
        S1C["Level 2: Frequency\nband powers, spectral\ncentroid, spectral entropy,\ndominant frequency"]:::sub
        S1D2["Level 2: Non-linear\nsample entropy, DFA\nalpha1, Poincare SD1/SD2"]:::sub
    end

    S1 --> O1[/"Per-signal feature DFs\n(one row per window)"/]:::out

    O1 --> S2["Step 2: Demographics Check\nValidate participant demographics\navailable from metadata"]:::proc

    S5 --> O5[/"encoded_features (per-signal)\nencoded_combined"/]:::out

    O5 --> S6["Step 6: FeatureSelector\nEnsemble ranking:\n(1) Mutual Information\n(2) Random Forest importance\n(3) F-statistic (ANOVA)\n(4) Variance threshold\nAverage normalised ranks\n--> Select top N"]:::proc
    S6 --> O6[/"feature_importance.csv\ntop_N_features.csv\ntop_N_features_encoded.csv"/]:::out

    subgraph S7D ["Three Methods Compared"]

        S7A["PCA\n(unsupervised)\nRetain 95% variance"]:::sub
        S7B["PLS-VIP\n(supervised)\nVIP >= 1.0 = important"]:::sub
        S7C["Truncated SVD\n(no centring)\nCompare components"]:::sub
    end

    S7 --> OUTPUT

    OUTPUT[/"OUTPUTS --> M5 Training (Planned):\n--- features_raw/ ---\n*_features.csv, combined_features.csv\n--- features_normalised/ ---\n*_features_norm.csv, combined_features_norm.csv\n--- features_encoded/ ---\n*_features_encoded.csv\ncombined_features_encoded.csv\n--- features_selected/ ---\nfeature_importance.csv\ntop_N_features.csv, top_N_features_encoded.csv\n--- dimensionality_reduction/ ---\npca_components.csv, pca_transformed.csv\npca_explained_variance.csv\npls_vip_scores.csv, svd_transformed.csv\ncomparison_report.json\n--- fitted_scaler.joblib ---\n--- feature_engineering_metadata.json ---"/]:::out

    HEAD[/"MODULE 5 — Data Analyser v1.0.0"\]:::head
    HEAD --> INPUT[/"Input:\nM3 output folder (signals +\nfeatures_raw/) | Dict of DFs"/]:::data

    INPUT --> S1["Step 1: Data Loading\nLoad signal CSVs (EDA...ACC)\n+ feature CSVs from features_raw/\n+ combined features\nFallback: reconstruct from\nfeature means if raw unavailable"]:::proc
    S1 --> O1[/"signals_dict\nfeatures_dict\ncombined_df"/]:::out

    O1 --> S2["Step 2: SignalAnalyser\nTemporal Dynamics Analysis"]:::proc
    subgraph S2D ["Event-Level Analysis"]
        direction TB
        S2A["Extract per-event segments\nfrom annotated signals\n(event_id, target_label,\nstart_s, end_s)"]:::sub
        S2A --> S2B["Per event x signal:\nBaseline: 30s pre-event\nmedian + std"]:::sub
        S2B --> S2C["Peak detection:\npeak value, time-to-peak,\n% change from baseline,\npeak alignment with event"]:::sub
        S2C --> S2D2["Recovery analysis:\ntime to subside (50% return)\nreturn to baseline check\npost-event median drift"]:::sub
        S2D2 --> S2E["Adaptive threshold detection:\nrunning median +/- N x MAD\n(300s window, 3x MAD,\nmin 30s sustained)"]:::sub

    O2 --> S3["Step 3: Statistical Analyses"]:::proc
    subgraph S3D ["6 Statistical Tests"]
        direction TB
        S3A["Descriptive statistics:\nper-target mean, std, median,\nIQR, min, max, CV"]:::sub
        S3A --> S3B["Kruskal-Wallis H-test:\nnon-parametric across all targets\n+ eta-squared effect size"]:::sub
        S3B --> S3C["Pairwise Mann-Whitney U:\ntop features, all target pairs\nBonferroni corrected (alpha=0.05)"]:::sub
        S3C --> S3D2["Spearman correlation:\nfeatures vs ordinal-encoded\ntarget label"]:::sub
        S3D2 --> S3E["Feature-feature correlation:\nSpearman matrix, top 30\nfeatures by variance"]:::sub
        S3E --> S3F["Signal % change summary:\nmean/median/min/max\nper signal x target"]:::sub
    end
    S3 --> O3[/"descriptive_df\nkruskal_df\npairwise_df\ncorr_target_df\ncorr_matrix_df\npct_change_df"/]:::out

        direction TB
        S4A["1. Descriptive boxplots\n(per signal, top features)"]:::sub
        S4A --> S4B["2. % change from baseline\n(bar charts per signal)"]:::sub
        S4B --> S4C["3. Temporal dynamics\n(time-to-peak, subside, return)"]:::sub
        S4C --> S4D2["4. Return-to-median rate\n5. Kruskal-Wallis features"]:::sub
        S4D2 --> S4E["6. Feature-target correlation\n7. Feature-feature heatmap"]:::sub

    end
    S4 --> O4[/"10+ PNG plots\n3d_signals.html (Plotly)"/]:::out
